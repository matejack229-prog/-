                target.style.opacity = '1';
                target.style.transform = 'translateY(0)';
            }, 50);

            // 更新页脚显示
            const footer = document.getElementById('global-footer');
            if (pageId === 'home') footer.classList.remove('hide');
            else footer.classList.add('hide');

            currentSection = pageId;
            window.scrollTo(0, 0);
        }

        function toggleMode() {
            const switchEl = document.getElementById('modeSwitch');
            const proElements = document.querySelectorAll('.pro-only');
            const label = document.getElementById('modeLabel');

            if (currentMode === 'Standard') {
                currentMode = 'Pro';
                switchEl.classList.add('active');
                label.innerText = 'Standard'; // 保持样式一致，实际上是通过高亮区分
                proElements.forEach(el => el.classList.remove('hide'));
            } else {
                currentMode = 'Standard';
                switchEl.classList.remove('active');
                proElements.forEach(el => el.classList.add('hide'));
            }
        }

        function startProcessing() {
            navigate('processing');
            const fill = document.getElementById('progress-fill');
            const text = document.getElementById('progress-text');
            const percent = document.getElementById('progress-percent');
            
            const stages = [
                { p: 25, t: '正在智能识别高光时刻...' },
                { p: 50, t: 'AI 正在同步音频卡点...' },
                { p: 75, t: '渲染 Liquid Glass 滤镜...' },
                { p: 100, t: '合成地图与字幕...' }
            ];

            let stageIdx = 0;
            const interval = setInterval(() => {
                if (stageIdx < stages.length) {
                    fill.style.width = stages[stageIdx].p + '%';
                    text.innerText = stages[stageIdx].t;
                    percent.innerText = stages[stageIdx].p + '%';
                    stageIdx++;
                } else {
                    clearInterval(interval);
                    setTimeout(() => navigate('preview'), 1000);
                }
            }, 1200);
        }

        // 初始化
        window.onload = () => {
            navigate('home');
        };
